cmake_minimum_required(VERSION 3.30)
project(HabitTrackerTDD LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.9.1 EXACT REQUIRED COMPONENTS Core Test Sql Widgets)
enable_testing()
add_library(habit_logic src/HabitLogic.cpp src/HabitLogic.h)
target_include_directories(habit_logic PUBLIC src)
target_link_libraries(habit_logic PUBLIC Qt6::Core)
add_executable(logic_tests tests/LogicTests.cpp)
target_link_libraries(logic_tests PRIVATE habit_logic Qt6::Test)
add_test(NAME habit_logic COMMAND logic_tests)
add_library(habit_storage src/HabitStore.cpp src/HabitStore.h)
target_link_libraries(habit_storage PUBLIC habit_logic Qt6::Sql)
add_executable(database_tests tests/DatabaseTests.cpp)
target_link_libraries(database_tests PRIVATE habit_storage Qt6::Test)
add_test(NAME habit_database COMMAND database_tests)
add_library(habit_ui src/AppWindow.cpp src/AppWindow.h src/CompletionChart.cpp src/CompletionChart.h src/CompletionDelegate.cpp src/CompletionDelegate.h)
target_link_libraries(habit_ui PUBLIC habit_storage Qt6::Widgets)
add_executable(window_tests tests/WindowTests.cpp)
target_link_libraries(window_tests PRIVATE habit_ui Qt6::Test)
add_test(NAME habit_window COMMAND window_tests)
add_executable(habit_tracker WIN32 src/main.cpp)
target_link_libraries(habit_tracker PRIVATE habit_ui)
add_executable(storage_probe tests/StorageProbe.cpp)
target_link_libraries(storage_probe PRIVATE habit_storage)
